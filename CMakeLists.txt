cmake_minimum_required(VERSION 3.22)
project(aster VERSION 0.1.0 LANGUAGES CXX)

set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
include(GNUInstallDirs)

find_package(Qt6 6.6 REQUIRED COMPONENTS Gui Quick Qml DBus Widgets)
find_package(LayerShellQt REQUIRED)
find_package(KF6WindowSystem REQUIRED)
find_package(KF6Service REQUIRED)
find_package(KF6KIO REQUIRED)
find_package(PolkitQt6-1 REQUIRED)
# Biblioteka powiadomień z plasma-workspace — wprost (jej pakiet CMake ciągnie pół KDE do zbudowania)
find_library(NOTIFICATIONMANAGER_LIBRARY NAMES notificationmanager REQUIRED)
find_path(NOTIFICATIONMANAGER_INCLUDE_DIR notificationmanager/server.h REQUIRED)
add_library(AsterNotificationManager INTERFACE)
target_link_libraries(AsterNotificationManager INTERFACE ${NOTIFICATIONMANAGER_LIBRARY})
target_include_directories(AsterNotificationManager INTERFACE ${NOTIFICATIONMANAGER_INCLUDE_DIR})

qt_standard_project_setup(REQUIRES 6.6)
qt_policy(SET QTP0004 NEW)
# Wszystkie moduły QML w jednym miejscu — narzędzia (qmllint) widzą wtedy Aster.Ui z każdej aplikacji
set(QT_QML_OUTPUT_DIRECTORY ${CMAKE_BINARY_DIR}/qml)

# Wspólne nagłówki C++ (np. typography.h) dla powłoki i aplikacji
include_directories(${CMAKE_SOURCE_DIR}/common)
add_subdirectory(common)
add_subdirectory(shell)
add_subdirectory(apps/files)
add_subdirectory(apps/browser)
add_subdirectory(apps/webapps)
add_subdirectory(apps/store)
add_subdirectory(apps/settings)
add_subdirectory(apps/terminal)
add_subdirectory(apps/tasks)
add_subdirectory(apps/photos)
add_subdirectory(apps/pdf)
add_subdirectory(apps/portal)
